; Native quest waypoints use one exterior door reference per transition.
; Keep this display list separate from the teleport landing coordinates for travel.
Begin Function {array_var aPathPoints}
    array_var aDisplayPoints = Ar_Construct "array"
    foreach array_var aEntry <- aPathPoints
        array_var aPoint = aEntry["value"]
        if eval aPoint["kind"] == "arrival"
            continue
        endif
        if eval aPoint["kind"] == "exit"
            ; Prefer the departure door. For an interior departure, use the
            ; linked exterior door reference carried by the next arrival point.
            if eval aPoint["worldspace"] == 0
                int iArrival = aEntry["key"] + 1
                if eval iArrival >= (Ar_Size aPathPoints)
                    continue
                endif
                let aPoint := aPathPoints[iArrival]
                if eval aPoint["kind"] != "arrival" || aPoint["worldspace"] == 0
                    continue
                endif
                ref rDoor = aPoint["reference"]
                if rDoor == 0
                    continue
                endif
                array_var aDoorPoint = Ar_Construct "stringmap"
                let aDoorPoint["kind"] := "exit"
                let aDoorPoint["reference"] := rDoor
                let aDoorPoint["worldspace"] := aPoint["worldspace"]
                let aDoorPoint["cell"] := aPoint["cell"]
                let aDoorPoint["x"] := rDoor.GetPos X
                let aDoorPoint["y"] := rDoor.GetPos Y
                let aDoorPoint["z"] := rDoor.GetPos Z
                let aPoint := aDoorPoint
            endif
        endif
        Ar_Append aDisplayPoints aPoint
    loop
    SetFunctionValue aDisplayPoints
End
