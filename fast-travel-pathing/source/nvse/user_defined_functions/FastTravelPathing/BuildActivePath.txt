; coordinates, space and heading belong to FTP.
Begin Function {array_var aPath, ref rFinalMarker}
    SetFunctionValue 0
    array_var aPoints = aPath["points"]
    if eval (Ar_Size aPoints) < 2
        return
    endif
    array_var aLast = aPoints[(Ar_Size aPoints) - 1]
    if eval aLast["kind"] != "destination" || rFinalMarker == 0
        return
    endif
    array_var aTravelPoints = Ar_Construct "array"
    array_var aArrivals = Ar_Construct "array"
    array_var aWorlds = Ar_Construct "array"
    array_var aCells = Ar_Construct "array"
    array_var aTimes = Ar_Construct "array"
    array_var aX = Ar_Construct "array"
    array_var aY = Ar_Construct "array"
    array_var aZ = Ar_Construct "array"
    array_var aHeading = Ar_Construct "array"
    ; Sample timing inputs once when travel is confirmed. GetPath is geometry only.
    float fSpeed = Player.GetRunSpeed
    float fScale = TimeScale
    int iValid = 1
    foreach array_var aEntry <- aPoints
        array_var aPoint = aEntry["value"]
        string_var sKind = aPoint["kind"]
        ref rPoint = 0
        float fHeading = 0
        array_var aArrival = Ar_Null
        if eval sKind == "exit"
            int iNext = aEntry["key"] + 1
            if eval iNext >= (Ar_Size aPoints)
                let iValid := 0
                break
            endif
            let aArrival := aPoints[iNext]
            if eval aArrival["kind"] != "arrival"
                let iValid := 0
                break
            endif
            let rPoint := aPoint["reference"]
        elseif eval sKind == "destination"
            let rPoint := rFinalMarker
            if eval Ar_HasKey aPoint "marker"
                let rPoint := aPoint["marker"]
            endif
            let aArrival := aPoint
        else
            continue
        endif
        if eval rPoint == 0 || aArrival["reference"] == 0
            let iValid := 0
            break
        endif
        if eval sKind == "exit"
            let fHeading := rPoint.GetDoorTeleportRot
        else
            ref rLanding = aArrival["reference"]
            let fHeading := rLanding.GetAngle Z
        endif
        int iPrevious = aEntry["key"] - 1
        if iPrevious < 0
            let iValid := 0
            break
        endif
        array_var aPrevious = aPoints[iPrevious]
        float fTime = Call (CompileScript "FastTravelPathing\GetSegmentTravelTime.txt") aPrevious aPoint fSpeed fScale
        if eval fTime < 0 || (fTime - fTime) != 0
            let iValid := 0
            break
        endif
        ref rCell = aArrival["cell"]
        ref rWorld = aArrival["worldspace"]
        if rWorld
            ; Resolve and store the landing CELL, including nonresident records,
            ; without loading its references.
            ; Missing cells stay unresolved and retain their travel points.
            ; GetExteriorCellEx takes grid indices; floor before assigning integers.
            int iGridX = Floor (aArrival["x"] / 4096)
            int iGridY = Floor (aArrival["y"] / 4096)
            let rCell := GetExteriorCellEx rWorld iGridX iGridY
        elseif rCell == 0
            ref rArrivalRef = aArrival["reference"]
            let rCell := rArrivalRef.GetParentCell
        endif
        Ar_Append aTimes fTime
        Ar_Append aTravelPoints rPoint
        Ar_Append aArrivals (aArrival["reference"])
        Ar_Append aWorlds (aArrival["worldspace"])
        Ar_Append aCells rCell
        Ar_Append aX (aArrival["x"])
        Ar_Append aY (aArrival["y"])
        Ar_Append aZ (aArrival["z"])
        Ar_Append aHeading fHeading
    loop
    if eval iValid == 0 || (Ar_Size aTravelPoints) == 0
        return
    endif
    Player.AuxVarSetFromArr "*_FTP_PathPoints" aTravelPoints
    Player.AuxVarSetFromArr "*_FTP_PathArrivals" aArrivals
    Player.AuxVarSetFromArr "*_FTP_PathWorlds" aWorlds
    Player.AuxVarSetFromArr "*_FTP_PathCells" aCells
    Player.AuxVarSetFromArr "*_FTP_PathTimes" aTimes
    Player.AuxVarSetFromArr "*_FTP_PathX" aX
    Player.AuxVarSetFromArr "*_FTP_PathY" aY
    Player.AuxVarSetFromArr "*_FTP_PathZ" aZ
    Player.AuxVarSetFromArr "*_FTP_PathHeading" aHeading
    Player.AuxVarSetFlt "*_FTP_PathIndex" -1
    Player.AuxVarSetFlt "*_FTP_PathWaitFrames" 0
    SetFunctionValue 1
End
