; Prepare all destination paths for one origin using a single Dijkstra search.
; Retained in the session cache by origin ID; graph/config changes reload the cache.
Begin Function {array_var aGraph, string_var sOriginLocationID}
    array_var aSearch = Call (CompileScript "FastTravelPathing\GetShortestPaths.txt") aGraph sOriginLocationID
    array_var aDistances = aSearch["distances"]
    array_var aPaths = Ar_Construct "stringmap"
    array_var aBest = Ar_Construct "stringmap"
    foreach array_var aEntry <- aDistances
        string_var sDestination = aEntry["key"]
        array_var aVertex = aGraph[sDestination]
        ref rCenter = aVertex["center"]
        string_var sMarkerID = GetFormIDString rCenter
        ; Shared centers represent several vertices. Keep the cheapest reachable
        ; destination, matching the previous hover selection behavior.
        if eval Ar_HasKey aBest sMarkerID
            if eval aEntry["value"] >= aBest[sMarkerID]
                continue
            endif
        endif
        let aBest[sMarkerID] := aEntry["value"]
        let aPaths[sMarkerID] := Call (CompileScript "FastTravelPathing\GetShortestPath.txt") aSearch sDestination
    loop
    array_var aResult = Ar_Construct "stringmap"
    let aResult["search"] := aSearch
    let aResult["paths"] := aPaths
    SetFunctionValue aResult
End
