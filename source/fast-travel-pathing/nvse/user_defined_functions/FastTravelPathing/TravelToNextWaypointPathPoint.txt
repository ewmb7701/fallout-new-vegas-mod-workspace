; caller supplies space, coordinates, degrees and game hours.
Begin Function {}
    int iAirBeforeCheck = Player.IsInAir
    if eval (GetPCCanFastTravelWithMessage) == 0
        WriteStringToFile "FastTravelPathing_Debug.log" 1 "ELIGIBILITY rejected air-before-check=%g%r" iAirBeforeCheck
        Call (CompileScript "FastTravelPathing\TraceTravel.txt") "STOP native-eligibility"
        Call (CompileScript "FastTravelPathing\ClearActiveWaypointPath.txt")
        return
    endif
    array_var aNext = Call (CompileScript "FastTravelPathing\GetNextWaypointPathPoint.txt")
    if eval aNext == Ar_Null
        Call (CompileScript "FastTravelPathing\TraceTravel.txt") "STOP no-next-point"
        Call (CompileScript "FastTravelPathing\ClearActiveWaypointPath.txt")
        return
    endif
    int iNext = aNext["index"]
    float fTime = aNext["time"]
    ref rPoint = Player.AuxVarGetRef "*_FTP_WaypointPathPoints" iNext
    ref rArrival = Player.AuxVarGetRef "*_FTP_WaypointPathArrivals" iNext
    WriteStringToFile "FastTravelPathing_Debug.log" 1 "NEXT index=%g point=%i (%n) arrival=%i hours=%g%r" iNext rPoint rPoint rArrival fTime
    if eval (Call (CompileScript "FastTravelPathing\CanTravelToWaypointPathPoint.txt") rPoint rArrival) == 0
        Call (CompileScript "FastTravelPathing\TraceTravel.txt") "STOP waypoint-validation"
        Call (CompileScript "FastTravelPathing\ClearActiveWaypointPath.txt")
        return
    endif
    ref rSpace = Player.AuxVarGetRef "*_FTP_WaypointPathWorlds" iNext
    if rSpace == 0
        let rSpace := Player.AuxVarGetRef "*_FTP_WaypointPathCells" iNext
    endif
    float fX = Player.AuxVarGetFlt "*_FTP_WaypointPathX" iNext
    float fY = Player.AuxVarGetFlt "*_FTP_WaypointPathY" iNext
    float fZ = Player.AuxVarGetFlt "*_FTP_WaypointPathZ" iNext
    float fHeading = Player.AuxVarGetFlt "*_FTP_WaypointPathHeading" iNext
    ; State precedes execution: native loading can run script callbacks.
    Player.AuxVarSetFlt "*_FTP_WaypointPathIndex" iNext
    Player.AuxVarSetFlt "*_FTP_WaypointPathExecuting" 1
    Player.AuxVarSetFlt "*_FTP_WaypointPathWaitFrames" 0
    if eval (FastTravelEx rSpace fX fY fZ 0 0 fHeading fTime) == 0
        Call (CompileScript "FastTravelPathing\TraceTravel.txt") "STOP FastTravelEx-failed"
        Call (CompileScript "FastTravelPathing\ClearActiveWaypointPath.txt")
        return
    endif
    Call (CompileScript "FastTravelPathing\TraceTravel.txt") "FastTravelEx-returned-success"
    if (Player.AuxVarSize "*_FTP_WaypointPathPoints") > 0
        Player.AuxVarSetFlt "*_FTP_WaypointPathExecuting" 0
    endif
End
