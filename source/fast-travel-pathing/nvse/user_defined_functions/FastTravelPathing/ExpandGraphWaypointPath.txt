; Keep each configured leg's native path and its endpoint references.
; The first leg starts at the player; subsequent legs start at marker landing refs.
; GetWaypointPath resolves door transitions but does not resolve marker linked refs for us.
Begin Function {array_var aGraph, array_var aIDs, ref rFinalMarker}
    SetFunctionValue Ar_Null
    if eval (Ar_Size aIDs) == 0
        return
    endif
    array_var aMarkers = Ar_Construct "array"
    if eval (Ar_Size aIDs) == 1
        ; Caller validated the explicit same-center connection before expansion.
        Ar_Append aMarkers rFinalMarker
    endif
    foreach array_var aID <- aIDs
        if eval aID["key"] > 0
            array_var aVertex = aGraph[aID["value"]]
            Ar_Append aMarkers (aVertex["center"])
        endif
    loop
    array_var aLegs = Ar_Construct "array"
    float fLength = 0
    int iValid = 1
    ref rOrigin = Player
    foreach array_var aEntry <- aMarkers
        ref rMarker = aEntry["value"]
        if eval rMarker == 0 || rMarker.GetDisabled || (rMarker.GetMapMarkerVisible == 0) || (rMarker.GetMapMarkerTravel == 0)
            let iValid := 0
            break
        endif
        ref rDestination = rMarker.GetLinkedRef
        if rDestination == 0
            let rDestination := rMarker
        endif
        array_var aLeg = GetWaypointPath rOrigin rDestination 2
        if eval aLeg == Ar_Null
            let iValid := 0
            break
        endif
        let fLength := fLength + aLeg["length"]
        array_var aRouteLeg = Ar_Construct "stringmap"
        let aRouteLeg["path"] := aLeg
        let aRouteLeg["originRef"] := rOrigin
        let aRouteLeg["destinationRef"] := rDestination
        let aRouteLeg["marker"] := rMarker
        Ar_Append aLegs aRouteLeg
        let rOrigin := rDestination
    loop
    if iValid == 0
        return
    endif
    array_var aResult = Ar_Construct "stringmap"
    let aResult["length"] := fLength
    let aResult["legs"] := aLegs
    SetFunctionValue aResult
End
