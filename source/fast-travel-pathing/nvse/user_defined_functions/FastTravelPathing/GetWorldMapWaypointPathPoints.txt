; Project travel coordinates using native marker anchors and reference samples.
Begin Function {array_var aDisplayPoints, array_var aLegs}
    array_var aPoints = Ar_Construct "stringmap"
    array_var aProjection = Ar_Construct "stringmap"
    string_var sMap = "MapMenu/GLOW_BRANCH/MM_MainRect/MM_WorldMap_ClipWindow/MM_WorldMap_ParentImage/"
    array_var aMarkers = GetFormRefs MapMarker
    foreach array_var aEntry <- aMarkers
        ref rCenter = aEntry["value"]
        string_var sName = rCenter.GetMapMarkerName
        if eval (Sv_Length sName) == 0
            continue
        endif
        string_var sTile = sMap + sName + ":0"
        if eval (GetUIFloatAlt (sTile + "/_MarkerIndex")) == -999
            continue
        endif
        ; Ambiguous named tiles cannot safely provide a center position.
        if eval (GetUIFloatAlt (sMap + sName + ":1/_MarkerIndex")) != -999
            continue
        endif
        float fX = GetUIFloatAlt (sTile + "/_x")
        float fY = GetUIFloatAlt (sTile + "/_y")
        if eval fX == -999 || fY == -999
            continue
        endif
        ref rWorld = rCenter.GetParentWorldspace
        string_var sWorld = GetFormIDString rWorld
        float fWorldX = rCenter.GetPos X
        float fWorldY = rCenter.GetPos Y
        if eval (Ar_HasKey aProjection sWorld) == 0
            array_var aAxis = Ar_Construct "stringmap"
            let aAxis["worldX"] := fWorldX
            let aAxis["worldY"] := fWorldY
            let aAxis["mapX"] := fX
            let aAxis["mapY"] := fY
            float fNativeX = -999
            float fNativeY = -999
            rCenter.GetWorldMapPosMults (fNativeX) (fNativeY)
            let aAxis["nativeX"] := fNativeX
            let aAxis["nativeY"] := fNativeY
            let aProjection[sWorld] := aAxis
        else
            let aAxis := aProjection[sWorld]
            if eval (Abs (fWorldX - aAxis["worldX"])) > 1
                let aAxis["scaleX"] := (fX - aAxis["mapX"]) / (fWorldX - aAxis["worldX"])
            endif
            if eval (Abs (fWorldY - aAxis["worldY"])) > 1
                let aAxis["scaleY"] := (fY - aAxis["mapY"]) / (fWorldY - aAxis["worldY"])
            endif
        endif
    loop
    ; The native cursor may provide a second calibration point in the player's
    ; exterior world when only one center is discovered. The player is also the first path endpoint.
    rWorld = Player.GetParentWorldspace
    sWorld = GetFormIDString rWorld
    if eval Ar_HasKey aProjection sWorld
        let aAxis := aProjection[sWorld]
        fWorldX = Player.GetPos X
        fWorldY = Player.GetPos Y
        fX = GetUIFloatAlt "MapMenu/GLOW_BRANCH/MM_MainRect/MM_WorldMap_ClipWindow/MM_WorldMapCursor/_x"
        fY = GetUIFloatAlt "MapMenu/GLOW_BRANCH/MM_MainRect/MM_WorldMap_ClipWindow/MM_WorldMapCursor/_y"
        if eval fX != -999 && fY != -999
            if eval (Ar_HasKey aAxis "scaleX") == 0 && (Abs (fWorldX - aAxis["worldX"])) > 1
                let aAxis["scaleX"] := (fX - aAxis["mapX"]) / (fWorldX - aAxis["worldX"])
            endif
            if eval (Ar_HasKey aAxis "scaleY") == 0 && (Abs (fWorldY - aAxis["worldY"])) > 1
                let aAxis["scaleY"] := (fY - aAxis["mapY"]) / (fWorldY - aAxis["worldY"])
            endif
        endif
    endif
    ; Sample both sides of every door, including departure references that are
    ; no longer displayed. They calibrate landing offsets in markerless worlds.
    array_var aSamples = Ar_Construct "array"
    foreach aEntry <- aLegs
        array_var aLeg = aEntry["value"]
        Ar_Append aSamples (aLeg["originRef"])
        Ar_Append aSamples (aLeg["destinationRef"])
        array_var aPath = aLeg["path"]
        foreach array_var aTransitionEntry <- (aPath["transitions"])
            array_var aTransition = aTransitionEntry["value"]
            ref rDoor = aTransition["door"]
            Ar_Append aSamples rDoor
            Ar_Append aSamples (rDoor.GetTeleportDoor)
        loop
    loop
    foreach aEntry <- aMarkers
        Ar_Append aSamples (aEntry["value"])
    loop
    foreach aEntry <- aSamples
        ref rSample = aEntry["value"]
        if rSample == 0
            continue
        endif
        rWorld = rSample.GetParentWorldspace
        if rWorld == 0
            continue
        endif
        sWorld = GetFormIDString rWorld
        if eval (Ar_HasKey aProjection sWorld) == 0
            ; Anchor a markerless child to its displayed ancestor, preserving
            ; JIP's child-world scale and offset in the native sample.
            ref rMapParent = GetWorldspaceParentWorldspace rWorld
            int iParentDepth = 0
            string_var sParent = ""
            while rMapParent && iParentDepth < 32
                sParent = GetFormIDString rMapParent
                if eval Ar_HasKey aProjection sParent
                    break
                endif
                let rMapParent := GetWorldspaceParentWorldspace rMapParent
                iParentDepth += 1
            loop
            if eval rMapParent == 0 || iParentDepth >= 32
                continue
            endif
            array_var aParentAxis = aProjection[sParent]
            fNativeX = -999
            fNativeY = -999
            rSample.GetWorldMapPosMults (fNativeX) (fNativeY)
            if eval fNativeX == -999 || fNativeY == -999 || aParentAxis["nativeX"] == -999 || aParentAxis["nativeY"] == -999
                continue
            endif
            let aAxis := Ar_Construct "stringmap"
            let aAxis["worldX"] := rSample.GetPos X
            let aAxis["worldY"] := rSample.GetPos Y
            let aAxis["nativeX"] := fNativeX
            let aAxis["nativeY"] := fNativeY
            let aAxis["mapX"] := fNativeX + aParentAxis["mapX"] - aParentAxis["nativeX"]
            let aAxis["mapY"] := fNativeY + aParentAxis["mapY"] - aParentAxis["nativeY"]
            let aProjection[sWorld] := aAxis
        endif
        let aAxis := aProjection[sWorld]
        if eval (Ar_HasKey aAxis "scaleX") && (Ar_HasKey aAxis "scaleY")
            continue
        endif
        fNativeX = -999
        fNativeY = -999
        rSample.GetWorldMapPosMults (fNativeX) (fNativeY)
        if eval fNativeX == -999 || fNativeY == -999 || aAxis["nativeX"] == -999 || aAxis["nativeY"] == -999
            continue
        endif
        fWorldX = rSample.GetPos X
        fWorldY = rSample.GetPos Y
        if eval (Ar_HasKey aAxis "scaleX") == 0 && (Abs (fWorldX - aAxis["worldX"])) > 1
            let aAxis["scaleX"] := (fNativeX - aAxis["nativeX"]) / (fWorldX - aAxis["worldX"])
        endif
        if eval (Ar_HasKey aAxis "scaleY") == 0 && (Abs (fWorldY - aAxis["worldY"])) > 1
            let aAxis["scaleY"] := (fNativeY - aAxis["nativeY"]) / (fWorldY - aAxis["worldY"])
        endif
    loop
    foreach aEntry <- aDisplayPoints
        array_var aPoint = aEntry["value"]
        array_var aSpace = aPoint["space"]
        array_var aPosition = aPoint["position"]
        let rWorld := aSpace["worldspace"]
        if rWorld == 0
            continue
        endif
        sWorld = GetFormIDString rWorld
        if eval (Ar_HasKey aProjection sWorld) == 0
            continue
        endif
        let aAxis := aProjection[sWorld]
        ; An axis already at its anchor needs no scale. In particular, a lone
        ; marker's own point must not disappear just because both scales are absent.
        fX = aAxis["mapX"]
        fY = aAxis["mapY"]
        if eval aPosition["x"] != aAxis["worldX"]
            if eval (Ar_HasKey aAxis "scaleX") == 0
                continue
            endif
            fX = fX + (aPosition["x"] - aAxis["worldX"]) * aAxis["scaleX"]
        endif
        if eval aPosition["y"] != aAxis["worldY"]
            if eval (Ar_HasKey aAxis "scaleY") == 0
                continue
            endif
            fY = fY + (aPosition["y"] - aAxis["worldY"]) * aAxis["scaleY"]
        endif
        string_var sPointIndex = $(aEntry["key"])
        let aPoints[sPointIndex] := Ar_List fX fY
    loop
    SetFunctionValue aPoints
End
