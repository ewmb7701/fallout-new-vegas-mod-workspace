; Unregister handlers and clear the owned connection lines.
Begin Function {}
    Call (CompileScript "FastTravelPathing\ClearActiveWaypointPath.txt")
    ref rOldUpdate = Player.AuxVarGetRef "*_FTP_rActiveWaypointPathUpdateHandler"
    if rOldUpdate
        Player.SetGameMainLoopCallback rOldUpdate 0
    endif
    Player.AuxVarErase "*_FTP_rActiveWaypointPathUpdateHandler"
    Player.AuxVarErase "*_FTP_ActiveWaypointPathMarkers"
    Player.AuxVarErase "*_FTP_ActiveWaypointPathIndex"
    ref rTravelBegin = Player.AuxVarGetRef "*_FTP_rBeforeFastTravelHandler"
    if rTravelBegin
        RemoveEventHandler "OnBeforeFastTravel" rTravelBegin
    endif
    Player.AuxVarErase "*_FTP_rBeforeFastTravelHandler"
    ref rTravel = Player.AuxVarGetRef "*_FTP_rFastTravelHandler"
    if rTravel
        SetOnFastTravelEventHandler rTravel 0
    endif
    Player.AuxVarErase "*_FTP_rFastTravelHandler"
    ref rLocationUpdateHandler = Player.AuxVarGetRef "*_FTP_rLocationUpdateHandler"
    if rLocationUpdateHandler
        Player.SetGameMainLoopCallback rLocationUpdateHandler 0
    endif
    Player.AuxVarErase "*_FTP_rLocationUpdateHandler"
    ref rHover = Player.AuxVarGetRef "*_FTP_rHoverHandler"
    if rHover
        RemoveEventHandler "OnWorldMapMenuSelectionChange" rHover
    endif
    Player.AuxVarErase "*_FTP_rHoverHandler"
    ref rBuild = Player.AuxVarGetRef "*_FTP_rBuildHandler"
    if rBuild
        RemoveEventHandler "OnWorldMapMenuBuild" rBuild
    endif
    ref rClose = Player.AuxVarGetRef "*_FTP_rCloseHandler"
    if rClose
        SetOnMenuCloseEventHandler rClose 0 1023
    endif
    Player.AuxVarErase "*_FTP_rBuildHandler"
    Player.AuxVarErase "*_FTP_rCloseHandler"
    Player.AuxVarErase "*_FTP_sPrevLocationId"
    Player.AuxVarErase "*_FTP_bPrevLocationValid"
    RemoveEventHandler "OnWorldMapMenuBuild" (CompileScript "FastTravelPathing\OnWorldMapMenuBuild.txt")
    SetOnMenuCloseEventHandler (CompileScript "FastTravelPathing\OnWorldMapClose.txt") 0 1023
    Call (CompileScript "FastTravelPathing\UnloadWorldMapSelectedWaypointPathUIComponents.txt")
End
