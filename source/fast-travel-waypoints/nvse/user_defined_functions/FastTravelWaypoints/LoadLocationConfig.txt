; Load a location ID and its spatial scope from a location INI.
Begin Function {string_var sFile}
    string_var sPath = "FastTravelWaypoints\Locations\" + sFile
    array_var aConfigScope = Call (CompileScript "FastTravelWaypoints\LoadLocationScopeConfig.txt") sPath
    array_var aConfigLocation = Ar_Construct "stringmap"
    let aConfigLocation["id"] := sFile[0:((Sv_Length sFile) - 5)]
    let aConfigLocation["scope"] := aConfigScope
    SetFunctionValue aConfigLocation
End
