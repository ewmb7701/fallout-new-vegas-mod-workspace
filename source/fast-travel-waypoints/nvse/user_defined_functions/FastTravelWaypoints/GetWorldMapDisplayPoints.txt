; Display the same coordinates and spaces consumed by BuildActiveWaypointPath.
; Interior slots remain in the sequence, but have no world-map projection.
Begin Function {array_var aLegs}
    array_var aPoints = Ar_Construct "array"
    foreach array_var aEntry <- aLegs
        array_var aLeg = aEntry["value"]
        array_var aPath = aLeg["path"]
        array_var aSpaces = aPath["spaces"]
        array_var aTransitions = aPath["transitions"]
        array_var aPoint = Ar_Null
        if eval (Ar_Size aPoints) == 0
            let aPoint := Ar_Construct "stringmap"
            let aPoint["position"] := aPath["origin"]
            let aPoint["space"] := aSpaces[0]
            Ar_Append aPoints aPoint
        endif
        foreach array_var aTransitionEntry <- aTransitions
            array_var aTransition = aTransitionEntry["value"]
            int iSpace = aTransitionEntry["key"] + 1
            let aPoint := Ar_Construct "stringmap"
            let aPoint["position"] := aTransition["arrival"]
            let aPoint["space"] := aSpaces[iSpace]
            Ar_Append aPoints aPoint
        loop
        let aPoint := Ar_Construct "stringmap"
        let aPoint["position"] := aPath["destination"]
        let aPoint["space"] := aSpaces[(Ar_Size aTransitions)]
        Ar_Append aPoints aPoint
    loop
    SetFunctionValue aPoints
End
