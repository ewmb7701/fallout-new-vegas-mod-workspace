; The configured directed graph is authoritative for reachability and marker order.
; Native paths expand the selected legs; they never introduce new graph connections.
Begin Function {ref rMarker, array_var aState}
    SetFunctionValue Ar_Null
    if eval rMarker == 0 || (rMarker.IsMapMarker == 0)
        return
    endif
    if eval rMarker.GetDisabled || (rMarker.GetMapMarkerVisible == 0) || (rMarker.GetMapMarkerTravel == 0)
        return
    endif
    array_var aConfig = aState["config"]
    array_var aLocations = aConfig["locations"]
    array_var aGraph = aState["graph"]
    string_var sOrigin = Call (CompileScript "FastTravelWaypoints\GetPlayerLocation.txt") aLocations
    array_var aCache = aState["graphWaypointPaths"]
    if eval (Ar_HasKey aCache sOrigin) == 0
        ; Outside configured scopes there is no cached route to select.
        return
    endif
    array_var aSearch = aCache[sOrigin]
    array_var aWaypointPaths = aSearch["paths"]
    string_var sMarker = GetFormIDString rMarker
    if eval (Ar_HasKey aWaypointPaths sMarker) == 0
        return
    endif
    array_var aEnabled = Call (CompileScript "FastTravelWaypoints\GetWorldMapMarkerEnabledStates.txt") aGraph sOrigin aWaypointPaths
    string_var sName = rMarker.GetMapMarkerName
    if eval (Ar_HasKey aEnabled sName) == 0
        return
    endif
    if eval aEnabled[sName] == 0
        return
    endif
    array_var aIDs = aWaypointPaths[sMarker]
    SetFunctionValue (Call (CompileScript "FastTravelWaypoints\ExpandGraphWaypointPath.txt") aGraph aIDs rMarker)
End
