; Return {index, time} for the next required arrival, including skipped leg times.
; Explicit policy: visit unvisited OR expired cells. Only visited, unexpired
; intermediate cells may be skipped. Unresolved/unvisited cells and the final point stay.
Begin Function {}
    SetFunctionValue Ar_Null
    int iIndex = (Player.AuxVarGetFlt "*_FTW_WaypointPathIndex") + 1
    int iCount = Player.AuxVarSize "*_FTW_WaypointPathPoints"
    if eval iIndex < 0 || iIndex >= iCount
        return
    endif
    float fTime = 0
    while iIndex < iCount
        float fLegTime = Player.AuxVarGetFlt "*_FTW_WaypointPathTimes" iIndex
        let fTime := fTime + fLegTime
        ref rCell = Player.AuxVarGetRef "*_FTW_WaypointPathCells" iIndex
        int iVisited = -1
        int iExpired = -1
        int iResolved = 0
        ref rWorld = Player.AuxVarGetRef "*_FTW_WaypointPathWorlds" iIndex
        if rWorld
            float fX = Player.AuxVarGetFlt "*_FTW_WaypointPathX" iIndex
            float fY = Player.AuxVarGetFlt "*_FTW_WaypointPathY" iIndex
            ; Query both fields once, including saved state for nonresident cells.
            int iGridX = Floor (fX / 4096)
            int iGridY = Floor (fY / 4096)
            array_var aCellState = GetExteriorCellState rWorld iGridX iGridY
            if eval aCellState != Ar_Null
                let iVisited := aCellState["visited"]
                let iExpired := aCellState["expired"]
                let iResolved := 1
            endif
        elseif rCell
            let iResolved := 1
            let iVisited := IsCellVisitedEx rCell
            let iExpired := IsCellExpiredEx rCell
        endif
        if eval iIndex == (iCount - 1) || iResolved == 0 || iVisited != 1 || iExpired > 0
            break
        endif
        ; Seen-data established a visit. Expiry -1 means no detach timestamp,
        ; not an expired cell; do not force another load for this case.
        let iIndex := iIndex + 1
    loop
    array_var aNext = Ar_Construct "stringmap"
    let aNext["index"] := iIndex
    let aNext["time"] := fTime
    SetFunctionValue aNext
End
